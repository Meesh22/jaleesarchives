    // تحميل البيانات من localStorage بالمفتاح المحدث jalees_archive_v14
    function loadArchiveData() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) {
          const parsed = JSON.parse(raw);
          if (Array.isArray(parsed) && parsed.length === 69) {
            return parsed;
          }
        }
      } catch (e) {
        console.error('Error reading localStorage:', e);
      }
      const initialData = buildInitialArchiveData();
      saveArchiveData(initialData);
      return initialData;
    }

    }

    // عرض بطاقات الأعضاء في الشبكة
    function renderMembersGrid() {
      const grid = document.getElementById('membersGrid');
      const emptyState = document.getElementById('emptyState');
      const counter = document.getElementById('resultsCounter');
      const q = currentSearchQuery.trim().toLowerCase();

      const filtered = archiveMembers.filter(member => {
        // التصفية حسب الفئة
        if (currentFilter === 'articles' && member.articles.length === 0) return false;
        if (currentFilter === 'quotes' && member.quotes.length === 0) return false;
        if (currentFilter === 'active' && (member.articles.length + member.quotes.length) < 10) return false;

        // التصفية حسب البحث النصي
        if (!q) return true;
        if (member.name.toLowerCase().includes(q)) return true;
        const inArticles = member.articles.some(a => a.title.toLowerCase().includes(q) || a.text.toLowerCase().includes(q));
        if (inArticles) return true;
        const inQuotes = member.quotes.some(qt => qt.text.toLowerCase().includes(q));
        return inQuotes;
      });

      counter.textContent = filtered.length === archiveMembers.length
        ? `يعرض جميع الأعضاء (${archiveMembers.length} عضواً)`
        : `نتائج التصفية: ${filtered.length} من أصل ${archiveMembers.length} عضواً`;

      if (filtered.length === 0) {
        grid.innerHTML = '';
        emptyState.classList.remove('hidden');
        return;
      }

      emptyState.classList.add('hidden');
      grid.innerHTML = filtered.map(member => {
        const articlesCount = member.articles.length;
        const quotesCount = member.quotes.length;
        const previewText = articlesCount > 0
          ? member.articles[0].title
          : (quotesCount > 0 ? member.quotes[0].text : 'عضو معتمد في نادي جليس - أصدقاء القراءة');

        return `
          <article
            onclick="openMemberModal(${member.id})"
            class="group bg-white/95 hover:bg-white rounded-2xl border border-[#C4A449]/35 hover:border-[#35746C] p-5 shadow-xs hover:shadow-md transition-all duration-200 flex flex-col justify-between cursor-pointer relative overflow-hidden"
          >

            <div>
              <div class="flex items-start justify-between gap-3 mb-3">
                <span class="inline-flex items-center justify-center w-8 h-8 rounded-xl bg-[#F7F4EC] border border-[#C4A449]/40 text-xs font-extrabold text-[#35746C] group-hover:bg-[#35746C] group-hover:text-white transition-colors shrink-0">
                  ${member.id}
                </span>
                <div class="flex items-center gap-1.5 flex-wrap justify-end">
                  ${articlesCount > 0 ? `
                    <span class="inline-flex items-center gap-1 bg-[#1A3A34] text-[#E8D8A6] text-[11px] font-bold px-2.5 py-0.5 rounded-full">
                      ${articlesCount} ${articlesCount === 1 ? 'مقال' : 'مقالات'}

                  ` : ''}

    function submitNewEntry(event) {
      event.preventDefault();
      const memberId = parseInt(document.getElementById('entryMemberSelect').value, 10);
      const type = document.getElementById('entryTypeSelect').value;
      const title = document.getElementById('entryTitleInput').value.trim();
      const text = document.getElementById('entryTextInput').value.trim();

      if (!text) return;

      const member = archiveMembers.find(m => m.id === memberId);
      if (!member) return;

      if (type === 'article') {
        member.articles.unshift({
          id: `a-${memberId}-${Date.now()}`,
          title: title || `خاطرة ومشاركة — ${member.name}`,
          text: text
        });
      } else {
        member.quotes.unshift({
          id: `q-${memberId}-${Date.now()}`,
          text: `«${text}» — ${member.name}`
        });
      }

      saveArchiveData(archiveMembers);
      updateStats();
      renderMembersGrid();
      closeAddModal();

      if (activeMemberId === memberId) {
        openMemberModal(memberId);
      }

      showToast('تمت إضافة المشاركة وحفظها في الأرشيف بنجاح');
    }

        };
        const blob = new Blob([JSON.stringify(exportPayload, null, 2)], { type: 'application/json;charset=utf-8' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `jalees-archive-69-members-${new Date().toISOString().slice(0, 10)}.json`;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
        showToast('تم تصدير ملف الأرشيف JSON بنجاح');
      } catch (e) {
        showToast('تعذر تصدير الملف');
      }
    }

    // نسخ النص إلى الحافظة
    function copyTextToClipboard(text) {
      const textarea = document.createElement('textarea');
      textarea.value = text;
      textarea.style.position = 'fixed';
      textarea.style.opacity = '0';
      document.body.appendChild(textarea);
      textarea.select();
      try {
        document.execCommand('copy');
        showToast('تم نسخ النص إلى الحافظة بنجاح');
      } catch (err) {
        showToast('حدث خطأ أثناء النسخ');
      }
      document.body.removeChild(textarea);
    }

    // إشعار Toast
    let toastTimer = null;
    function showToast(message) {
      const box = document.getElementById('toastBox');
      const msgEl = document.getElementById('toastMessage');
      msgEl.textContent = message;
      box.classList.remove('hidden');
      if (toastTimer) clearTimeout(toastTimer);
      toastTimer = setTimeout(() => {
        box.classList.add('hidden');
      }, 2800);
    }

    // حماية النصوص من XSS
    function escapeHTML(str) {
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }
